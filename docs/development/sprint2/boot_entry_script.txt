######### How to set TDCR according Docs ########
### set to TDCR rate:
```bash
echo "0x800" | sudo tee /sys/devices/platform/bus@0/c310000.mttcan/net/can0/tdc_offset;
echo "0x800" | sudo tee /sys/devices/platform/bus@0/c320000.mttcan/net/can1/tdc_offset;
```

### check:
```bash
cat /sys/devices/platform/bus@0/c310000.mttcan/net/can0/tdc_offset
cat /sys/devices/platform/bus@0/c320000.mttcan/net/can1/tdc_offset 
```

### set up can0/can1 device
script:
```bash
cd ~/workspace/agx_arm_ros/scripts/ && sudo ./activate_native_can.sh 
```

or manual:
```bash
sudo ip link set can0 type can   bitrate 1000000 sample-point 0.8   dbitrate 5000000 dsample-point 0.8   fd on loopback off restart-ms 100 berr-reporting on one-shot on
sudo ip link set can0 type can   bitrate 1000000 sample-point 0.8   dbitrate 5000000 dsample-point 0.8   fd on loopback off restart-ms 100 berr-reporting on one-shot on
```

  python3 - "$V" <<'PY'
from pathlib import Path
import re
import sys

value = sys.argv[1].lower()
p = Path(f"tegra234_can_tdcr_{value}.dts")
s = p.read_text()

# IMPORTANT:
# Patch ONLY the CAN0 5 Mbps prod-setting:
#   mttcan@c310000 -> prod_c_can_5m -> board -> prod
#
# Do NOT replace the first matching
#   prod = <0x00 0x48 0x7f00 ...>
# entry globally, because that can accidentally patch prod_c_can_2m_1m
# while leaving prod_c_can_5m stuck at the old value.
pattern = re.compile(
    r'(mttcan@c310000\s*\{.*?'
    r'prod_c_can_5m\s*\{\s*'
    r'board\s*\{\s*'
    r'prod\s*=\s*)'
    r'<0x00\s+0x48\s+0x7f00\s+0x[0-9a-fA-F]+>'
    r'(;\s*\};\s*\};)',
    re.S
)

replacement = rf'\1<0x00 0x48 0x7f00 0x{value}>\2'

s2, n = pattern.subn(replacement, s, count=1)

if n != 1:
    raise SystemExit(
        f"Expected exactly one CAN0 prod_c_can_5m entry to patch, patched {n}. "
        "Check that the DTS contains mttcan@c310000/prod_c_can_5m."
    )

p.write_text(s2)
print(f"Patched CAN0 prod_c_can_5m to 0x{value}")
PY

  dtc -I dts -O dtb \
    -o tegra234_can_tdcr_${V}.dtb \
    tegra234_can_tdcr_${V}.dts

  sudo cp tegra234_can_tdcr_${V}.dtb /boot/dtb/
done

sync
